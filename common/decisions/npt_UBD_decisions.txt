			has_country_flag = UBD_education_germanised
		}
		ai_will_do = {
			factor = 1
		}
		days_remove = 50
		remove_effect = {
			add_research_slot =  1
			if = {
				limit = { has_idea = no_literacy }
				swap_ideas = {
					remove_idea = no_literacy
					add_idea = cities_low_literacy
				}
			}
			else_if = {
				limit = { has_idea = cities_low_literacy }
				swap_ideas = {
					remove_idea = cities_low_literacy

				}

			else_if = {
				limit = { has_idea = cities_averange_literacy }

				limit = { has_idea = cities_averange_education }

					remove_idea = cities_averange_education

				}
			}
			else_if = {
				limit = { has_idea = low_literacy_in_countryside }
				swap_ideas = {
					remove_idea = low_literacy_in_countryside
					add_idea = mass_literacy
				}
			}
			else_if = {
				limit = { has_idea = mass_literacy }
				swap_ideas = {
					remove_idea = mass_literacy

				}
				set_country_flag = UBD_education_reform_completed
